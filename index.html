Sheets setup

  1. In Google Drive, create a sheet. Under Extensions, choose Apps Script.
  2. Paste the code below, then Deploy, New deployment, Web app. Set access to Anyone.
  3. Copy the web app URL into the box above.
function doPost(e) {
  var d = JSON.parse(e.postData.contents);
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  if (sh.getLastRow() === 0) {
    sh.appendRow(["timestamp","workout_date","exercise","set","weight_kg","reps","set_time","rest_seconds"]);
  }
  sh.appendRow([d.timestamp, d.workoutDate, d.exercise, d.set, d.weight, d.reps, d.setTime, d.restSeconds]);
  return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON);
}
